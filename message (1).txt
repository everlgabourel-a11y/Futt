-- FUT Hub complete source
local previous=getgenv().FUTHubBundle
if previous and previous.Cleanup then previous.Cleanup() end
local bundle={states={}}
getgenv().FUTHubBundle=bundle
function bundle.Cleanup()
 for i=#bundle.states,1,-1 do
  local s=bundle.states[i]
  s.live=false
  for _,conn in s.connections do pcall(function() conn:Disconnect() end) end
  for j=#s.cleanups,1,-1 do pcall(s.cleanups[j]) end
 end
 table.clear(bundle.states)
end
local function createState()
 local s={live=true,connections={},cleanups={}}
 local api={}
 function api.alive() return s.live end
 function api.connect(signal,fn)
  local conn=signal:Connect(fn)
  table.insert(s.connections,conn)
  return conn
 end
 function api.onCleanup(fn) table.insert(s.cleanups,fn) end
 table.insert(bundle.states,s)
 return api
end

   local pos=ball.Position

   samples[ball]={pos=pos,time=now,velocity=previous and previous.velocity}
   if previous then
    local elapsed=now-previous.time
    if elapsed>0.005 and elapsed<0.2 then
     local measured=(pos-previous.pos)/elapsed
     local oldVelocity=previous.velocity
     local velocity=oldVelocity and oldVelocity:Lerp(measured,0.55) or measured
     samples[ball].velocity=velocity
     if measured.Magnitude<450 and velocity.Magnitude>15 then

       if candidate:IsA("BasePart") then
        local d=(candidate.Position-root.Position).Magnitude

       end

       local normal=goal.CFrame.LookVector
       local toward=velocity:Dot(normal)
       if math.abs(toward)>12 then
        local goalTime=(goal.Position-pos):Dot(normal)/toward
        local keeperTime=(root.Position-pos):Dot(normal)/toward
        if goalTime>0 and goalTime<1.5 and keeperTime>0 and keeperTime<0.8 then
         local acceleration=Vector3.zero
         if oldVelocity then
          local ay=(velocity.Y-oldVelocity.Y)/elapsed
          acceleration=Vector3.new(0,math.clamp(ay,-workspace.Gravity,0),0)
         end
         local radius=ball.Size.Y*0.5
         local groundY=goal.Position.Y-goal.Size.Y*0.5+radius
         local filters={engine,character}
         for _,other in Players:GetPlayers() do
          if other.Character then table.insert(filters,other.Character) end
         end
         local params=RaycastParams.new()
         params.FilterType=Enum.RaycastFilterType.Exclude
         params.FilterDescendantsInstances=filters
         local ground=workspace:Raycast(pos+Vector3.new(0,2,0),Vector3.new(0,-14,0),params)
         if ground then groundY=ground.Position.Y+radius end
         local rolling=pos.Y<=groundY+0.9 and math.abs(velocity.Y)<6
         if rolling then acceleration=Vector3.zero; velocity=Vector3.new(velocity.X,0,velocity.Z) end
         local goalHit=pos+velocity*goalTime+acceleration*(0.5*goalTime*goalTime)
         goalHit=Vector3.new(goalHit.X,math.max(goalHit.Y,groundY),goalHit.Z)
         local localHit=goal.CFrame:PointToObjectSpace(goalHit)
         if math.abs(localHit.X)<goal.Size.X/2+1 and math.abs(localHit.Y)<goal.Size.Y/2+1 then
          local intercept=pos+velocity*keeperTime+acceleration*(0.5*keeperTime*keeperTime)
          intercept=Vector3.new(intercept.X,math.max(intercept.Y,groundY),intercept.Z)
          local offset=root.CFrame:VectorToObjectSpace(intercept-root.Position)
          local high=intercept.Y>root.Position.Y+1.2
          local lateral = math.abs(offset.X)
          local ballDistance = (pos-root.Position).Magnitude
          local needsJump = config.JumpDive and intercept.Y>root.Position.Y+config.JumpHeight
          local leadTime = config.MinLeadTime + (config.MaxLeadTime-config.MinLeadTime)*math.clamp(lateral/config.MaxDiveReach,0,1)
          if not high then
           leadTime=math.max(leadTime,config.LowLeadTime+0.06*math.clamp(lateral/config.MaxDiveReach,0,1))
          end
          if needsJump then leadTime+=config.JumpDelay end
          stats.prediction = {
           position = {x=intercept.X,y=intercept.Y,z=intercept.Z},
           seconds = keeperTime,
           distance = ballDistance,
           lateral = lateral,
           radius = config.Radius,
           ready = ballDistance<=config.Radius and keeperTime<=leadTime,
          }
          if ballDistance<=config.Radius and keeperTime<=leadTime and lateral<config.MaxDiveReach and intercept.Y<goal.Position.Y+goal.Size.Y/2+3 then
           if not best or keeperTime<best.time then
            best={time=keeperTime,right=offset.X>=0,high=high,jump=needsJump}
           end
          end
         end
        end
       end
      end
     end
    end
   end
  end

  lastDive=now
  stats.dives+=1

  if ball:IsA("BasePart") and ball.Name=="Ball" then
   local p=ball.Position
   local old=samples[ball]

   if old then
    local dt=now-old.t
    if dt>0.005 and dt<0.2 then
     local measured=(p-old.p)/dt

     samples[ball].v=v
     local delta=p-root.Position
     local horizontal=Vector3.new(delta.X,0,delta.Z)
     local relative=v-root.AssemblyLinearVelocity
     local hv=Vector3.new(relative.X,0,relative.Z)

     local approaching=horizontal:Dot(hv)<-4 or (horizontal.Magnitude<config.Reach and v.Y< -2)
     if measured.Magnitude<450 and approaching and delta.Magnitude<config.Radius then
      local eta
      if speed2>16 then
       eta=-horizontal:Dot(hv)/speed2
      else
       local fall=math.max(-v.Y,1)
       eta=math.max(0.03,(p.Y-(root.Position.Y+4))/fall)
      end
      local miss=(horizontal+hv*eta).Magnitude

       local grounded=humanoid.FloorMaterial~=Enum.Material.Air

       local jumpSpeed=humanoid.UseJumpPower and humanoid.JumpPower or math.sqrt(2*workspace.Gravity*humanoid.JumpHeight)
       local jumpRise=math.max(0,jumpSpeed*eta-0.5*workspace.Gravity*eta*eta)

       local expectedChest=chestY+(needsJump and jumpRise or root.AssemblyLinearVelocity.Y*eta)
       if targetY>root.Position.Y+0.6 and math.abs(targetY-expectedChest)<3 and eta<=(needsJump and config.JumpLead or config.LeadTime) then
        if not best or eta<best.eta then best={eta=eta,jump=needsJump,ball=ball} end
       end
      end
     end
    end
   end
  end
 end
 for ball in samples do if not ball.Parent then samples[ball]=nil end end
 if not stats.enabled or not valid() or now-lastAction<config.Cooldown or not best then return end

 stats.actions+=1

local function watch(ball)
 if not ball:IsA("BasePart") or ball.Name~="Ball" or watched[ball] then return end
 watched[ball]=true
 STATE.connect(ball.ChildAdded,function(child)
  if not child:IsA("BodyVelocity") then return end

   if not STATE.alive() or not stats.enabled or child.Name~="ShootVelocity" or os.clock()>armedUntil or not shooting() then return end

   local goal=goalName and goals:FindFirstChild(goalName)

   local target=goal.CFrame:PointToWorldSpace(Vector3.new(side*(goal.Size.X/2-2.5),goal.Size.Y/2-2.3,0))

   local horizontal=Vector3.new(delta.X,0,delta.Z)
   local speed=math.clamp(Vector3.new(child.Velocity.X,0,child.Velocity.Z).Magnitude,config.MinForwardSpeed,config.MaxForwardSpeed)
   local flight=math.max(horizontal.Magnitude/speed,0.15)
   local lift=0
   for _,force in ball:GetChildren() do
    if force:IsA("BodyForce") then lift+=force.Force.Y/math.max(ball.AssemblyMass,0.01) end
   end
   local gravity=math.clamp(workspace.Gravity-lift,0,workspace.Gravity)
   local vertical=math.clamp(delta.Y/flight+0.5*gravity*flight,0,speed*config.MaxLiftRatio)
   local velocity=horizontal/flight+Vector3.new(0,vertical,0)
   child.Velocity=velocity
   ball.AssemblyLinearVelocity=velocity
   pending[ball]={body=child,target=target,speed=speed,expires=os.clock()+0.5}
   armedUntil=0
   stats.shots+=1
   stats.last=side<0 and "Upper corner A" or "Upper corner B"
  end)
 end)
end

 end

local samples={}
local lastAttempt=0
local diveStarted=nil
local heldSince=nil
local lastHeld=nil

local function holding()
 local c=player.Character
 local carrier=c and c:FindFirstChild("Ball")
 local ref=carrier and carrier:FindFirstChild("ballobj")
 return carrier and carrier:FindFirstChild("catch") and ref and ref.Value~=nil
end
stats.IsHolding=holding
STATE.onCleanup(function() stats.enabled=false end)
STATE.connect(RunService.Heartbeat,function()
 local now=os.clock()
 if _G.Diving then
  if not diveStarted then diveStarted=now end
 else
  diveStarted=nil
 end
 local catchReady=not diveStarted or now-diveStarted>=0.28
 local held=holding()
 stats.holding=not not held
 if held then
  if not heldSince then heldSince=now end
  stats.heldSeconds=math.min(5,math.floor(now-heldSince)+1)
  stats.last="Holding "..tostring(stats.heldSeconds).."/5"
  local carrier=player.Character and player.Character:FindFirstChild("Ball")
  local ref=carrier and carrier:FindFirstChild("ballobj")
  lastHeld=ref and ref.Value
  return
 elseif heldSince then
  heldSince=nil
  stats.heldSeconds=0
  stats.last="Released"
 end
 local c=player.Character
 local root=c and c:FindFirstChild("HumanoidRootPart")
 local tool=c and c:FindFirstChild("GK")
 if not root then return end
 for _,ball in engine:GetChildren() do
  if ball:IsA("BasePart") and ball.Name=="Ball" then
   local pos=ball.Position
   local old=samples[ball]
   samples[ball]={p=pos,t=now}
   if stats.enabled and catchReady and tool and now-lastAttempt>1.0 and old and isrbxactive() and not UIS:GetFocusedTextBox() then
    local dt=now-old.t
    if dt>0.005 and dt<0.2 then
     local v=(pos-old.p)/dt-root.AssemblyLinearVelocity
     local delta=pos-root.Position
     local speed2=v:Dot(v)
     if speed2>100 and v.Magnitude<450 and delta.Magnitude<24 and delta:Dot(v)<20 then
      local eta=-delta:Dot(v)/speed2
      local predicted=delta+v*eta
      local localHit=root.CFrame:VectorToObjectSpace(predicted)
      if eta> -0.15 and eta<0.6 and predicted.Magnitude<13 then
       local client=tool:FindFirstChild("Client")
       local high=pos.Y+v.Y*eta>root.Position.Y+0.5
       local module=client and client:FindFirstChild(high and "catchhigh" or "catchlow")
       if module then
        lastAttempt=now
        stats.attempts+=1
        stats.last=high and "High catch attempt" or "Low catch attempt"
        task.spawn(function()
         local ok,err=pcall(function() require(module).new() end)
         if not ok then stats.last=tostring(err) end
        end)
       end
      end
     end
    end
   end
  end
 end
 for ball in samples do if not ball.Parent then samples[ball]=nil end end
end)
print("Independent auto catch active. Native five-second drop; no added pause.")

end

do
local STATE=createState()
local Players=game:GetService("Players")
local UIS=game:GetService("UserInputService")
local RunService=game:GetService("RunService")
local player=Players.LocalPlayer
local engine=workspace:WaitForChild("Game"):WaitForChild("Engine")
local config={OpponentRadius=10,BallRadius=7}
local stats={enabled=true,active=false,last="Waiting for nearby pressure"}
getgenv().FUTAutoShield=stats
getgenv().FUTAutoShieldConfig=config
local changedHumanoid=nil
local originalRotate=nil
local function release()
 if changedHumanoid and changedHumanoid.Parent then changedHumanoid.AutoRotate=originalRotate end
 changedHumanoid=nil
 stats.active=false
end
stats.SetEnabled=function(v) stats.enabled=v; if not v then release() end end
STATE.onCleanup(function() stats.enabled=false; release() end)
STATE.connect(RunService.Heartbeat,function(dt)
 local c=player.Character
 local root=c and c:FindFirstChild("HumanoidRootPart")
 local humanoid=c and c:FindFirstChildOfClass("Humanoid")
 local tool=c and c:FindFirstChildOfClass("Tool")
 if not stats.enabled or not root or not humanoid or not tool or string.lower(tool.Name)~="dribble" or _G.Diving or _G.Catching or not isrbxactive() or UIS:GetFocusedTextBox() or UIS:IsMouseButtonPressed(Enum.UserInputType.MouseButton1) then release(); return end
 local ball=nil
 for _,b in engine:GetChildren() do
  if b:IsA("BasePart") and b.Name=="Ball" then
   local owner=b:FindFirstChild("NetworkOwner")
   if owner and owner.Value==player and (b.Position-root.Position).Magnitude<config.BallRadius then ball=b; break end
  end
 end
 if not ball then release(); return end
 local opponent=nil
 local distance=config.OpponentRadius
 for _,other in Players:GetPlayers() do
  local team=other.Team and string.lower(other.Team.Name) or ""
  local ownTeam=player.Team and string.lower(player.Team.Name) or ""
  local sameSide=(string.find(team,"home") and string.find(ownTeam,"home")) or (string.find(team,"away") and string.find(ownTeam,"away"))
  if other~=player and not sameSide then
   local r=other.Character and other.Character:FindFirstChild("HumanoidRootPart")
   if r then local d=(r.Position-root.Position).Magnitude; if d<distance then distance=d; opponent=r end end
  end
 end
 if not opponent then release(); return end
 local away=root.Position-opponent.Position
 away=Vector3.new(away.X,0,away.Z)
 if away.Magnitude<0.1 then release(); return end
 if changedHumanoid~=humanoid then release(); changedHumanoid=humanoid; originalRotate=humanoid.AutoRotate end
 humanoid.AutoRotate=false
 local target=CFrame.lookAt(root.Position,root.Position+away.Unit)
 root.CFrame=root.CFrame:Lerp(target,1-math.exp(-12*dt))
 stats.active=true
 stats.last="Turning away from nearby opponent"
end)
print("Auto shield positioning active with Dribble. Releases for LMB actions.")

end

do
local STATE=createState()
local player=game:GetService("Players").LocalPlayer
local stats={enabled=true,applied=false}
local config={Reach=3}
getgenv().FUTFootReach=stats
getgenv().FUTFootReachConfig=config
local originals={}
local function restore()
 for foot,old in originals do foot.Scale=old.Scale; foot.Offset=old.Offset end
 table.clear(originals)
 stats.applied=false
end
local function update()
 if not stats.enabled then restore(); return end
 local character=player.Character
 local tool=(character and character:FindFirstChild("Dribble")) or player.Backpack:FindFirstChild("Dribble")
 local source=tool and tool:FindFirstChild("HitboxConfig")
 local physical=character and character:FindFirstChild("RightFoot")
 if not source or not physical then return end
 local cfg=require(source)
 local foot=cfg.Limbs and cfg.Limbs.Foot
 if not foot then return end
 if not originals[foot] then originals[foot]={Scale=foot.Scale,Offset=foot.Offset} end
 local size=physical.Size
 foot.Scale=Vector3.new(config.Reach/math.max(size.X,0.1),math.max(originals[foot].Scale.Y,1),config.Reach/math.max(size.Z,0.1))
 stats.applied=true
end
stats.SetEnabled=function(v) stats.enabled=v; update() end
STATE.onCleanup(function() stats.enabled=false; restore() end)
local elapsed=0
STATE.connect(game:GetService("RunService").Heartbeat,function(dt)
 elapsed+=dt
 if elapsed>=0.25 then elapsed=0; update() end
end)
update()
print("Foot reach active: 3-stud horizontal foot hitboxes for Dribble skills.")

end
